window = [(0, 3), (4, 3), (4, 0), (0, 0)]

xmin = window[0][0]
xmax = window[1][0]
ymin = window[3][1]
ymax = window[0][1]

print(f'xmin {xmin},  xmax {xmax},   ymin {ymin},   ymax{ymax}')

polygon = [(1, 5), (4, 5), (3, 1), (2, 1), (2, 4), (1, 1), (1, 5)]

subject_list = []

for i in range(len(polygon) - 1)
    p1 = polygon[i]
    p2 = polygon[i + 1]

    if (xmin = p1[0] = xmax and ymin = p1[1] = ymax) == (xmin = p2[0] = xmax and ymin = p2[1] = ymax)
        subject_list.append(p1)
    else
        subject_list.append(p1)
        subject_list.append([333])

print()
print('Subject_list', subject_list)
